// Load weather data on page load if (document.getElementById('weather')) { loadWeatherDashboard(); } // Weather Dashboard Loader async function loadWeatherDashboard() { const loading = document.getElementById('weatherLoading'); const content = document.getElementById('weatherContent'); const error = document.getElementById('weatherError'); try { const response = await fetch(`${API_URL}?action=getWeather`); const result = await response.json(); if (result.success && result.farms) { loading.style.display = 'none'; content.style.display = 'block'; renderWeatherCards(result.farms); } else { throw new Error(result.error || 'Failed to load weather data'); } } catch (err) { loading.style.display = 'none'; error.style.display = 'block'; error.textContent = 'Unable to load weather data: ' + err.message; } } function renderWeatherCards(farms) { const container = document.getElementById('weatherContent'); container.innerHTML = ''; farms.forEach(farmData => { if (!farmData.success) return; const { farm, weather } = farmData; const current = weather.current; const weatherCard = document.createElement('div'); weatherCard.style.cssText = 'background: white; border-radius: 10px; padding: 20px; margin-bottom: 15px; box-shadow: 0 2px 8px rgba(0,0,0,0.1);'; weatherCard.innerHTML = `

${farm.farmName}

${farm.location}, ${farm.city}

${current.temperature}°C
💧 Humidity
${current.humidity}%
🌧️ Precipitation
${current.precipitation} mm
💨 Wind Speed
${current.windSpeed} km/h
Last updated: ${new Date(current.time).toLocaleString('en-IN', { timeZone: 'Asia/Kolkata' })}
`; container.appendChild(weatherCard); }); }
